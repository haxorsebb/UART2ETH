# Transmission Debug Test
add_executable(transmission_debug_test
    transmission_debug_test.c
)

target_include_directories(transmission_debug_test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)

target_link_libraries(transmission_debug_test
    pico_stdlib
    hardware_spi
    hardware_gpio
    hardware_uart
    network_manager
)

# Enable USB and UART stdio
pico_enable_stdio_usb(transmission_debug_test 1)
pico_enable_stdio_uart(transmission_debug_test 1)

# Create UF2 output
pico_add_extra_outputs(transmission_debug_test)
